function expenseRows(xs){ return xs.slice().sort((a,b)=>a.date<b.date?-1:1).map(x=>({Datum:x.date, Omschrijving:x.desc, Bedrag:+x.amount||0, Betaling:x.pay==='cash'?'Cash':'Bancontact'})); }

function sheet(rows, cols, widths){ const ws=XLSX.utils.json_to_sheet(rows.length?rows:[], {header:cols}); ws['!cols']=widths.map(w=>({wch:w})); return ws; }

async function saveFile(filename, data){

  if(!S.dl){ toast('Downloaden is niet beschikbaar in deze weergave.',true); return false; }

  try{ await S.dl.save({filename, data}); return true; }

  catch(e){ if(e&&e.code==='declined') return false; toast('Het bestand kon niet bewaard worden. Probeer opnieuw.',true); return false; }

}

async function exportExcel(kind){

  if(typeof XLSX==='undefined'){ toast('Excel-module is niet geladen. Herlaad de pagina.',true); return; }

  const wb=XLSX.utils.book_new();

  const VC=['Datum','Klant','Behandelingen','Subtotaal','Korting','Totaal','Betaling','Opmerking'], VW=[12,22,48,10,9,10,12,30];

  const XC=['Datum','Omschrijving','Bedrag','Betaling'], XW=[12,40,10,12];

  let name;

  if(kind==='month'){

    const st=monthStats(S.ym);

    const sum=[['Sweet Nails by Thalyssa'],['Maandoverzicht', ymLabel(S.ym)],[],

      ['Inkomsten totaal',st.income],['  waarvan cash',st.cash],['  waarvan Bancontact',st.bank],['Korting gegeven',st.disc],

      ['Aantal bezoeken',st.visits],['Aantal klanten',st.clients],['Kosten',st.costs],['Netto',st.net]];

    const ws=XLSX.utils.aoa_to_sheet(sum); ws['!cols']=[{wch:24},{wch:16}];

    XLSX.utils.book_append_sheet(wb, ws, 'Samenvatting');

    XLSX.utils.book_append_sheet(wb, sheet(visitRows(S.visits[S.ym]||[]),VC,VW), 'Bezoeken');

    XLSX.utils.book_append_sheet(wb, sheet(expenseRows(S.expenses[S.ym]||[]),XC,XW), 'Kosten');

    name=`Sweet Nails ${ymLabel(S.ym)}.xlsx`;

  } else {

    const rows=[['Maand','Inkomsten','Cash','Bancontact','Bezoeken','Klanten','Kosten','Netto']];

    let allV=[], allX=[];

    for(let m=1;m<=12;m++){ const ym=S.year+'-'+pad(m), st=monthStats(ym); rows.push([MONTHS[m-1],st.income,st.cash,st.bank,st.visits,st.clients,st.costs,st.net]); allV=allV.concat(S.visits[ym]||[]); allX=allX.concat(S.expenses[ym]||[]); }

    const tot=rows.slice(1).reduce((a,r)=>a.map((x,i)=>i===0?'Totaal':x+(+r[i]||0)),['Totaal',0,0,0,0,0,0,0]); tot[5]=''; rows.push(tot.map(v=>typeof v==='number'?r2(v):v));

    const ws=XLSX.utils.aoa_to_sheet(rows); ws['!cols']=[{wch:12},{wch:11},{wch:10},{wch:11},{wch:9},{wch:8},{wch:10},{wch:10}];

    XLSX.utils.book_append_sheet(wb, ws, 'Per maand');

    XLSX.utils.book_append_sheet(wb, sheet(visitRows(allV),VC,VW), 'Bezoeken');

    XLSX.utils.book_append_sheet(wb, sheet(expenseRows(allX),XC,XW), 'Kosten');

    name=`Sweet Nails ${S.year}.xlsx`;

  }

  const buf=XLSX.write(wb,{type:'array',bookType:'xlsx'});

  if(await saveFile(name, new Blob([buf]))) toast('Excel-bestand bewaard');

}

function renderBackupInfo(){

  const el=$('backupInfo');

  el.innerHTML = S.backupAt ? `Laatste backup: <strong>${dateShort(S.backupAt.slice(0,10))}</strong>` : 'Nog geen backup gemaakt.';

  $('bkDown').hidden=!S.dl;

  const r=S.restore, box=$('restoreBox');

  if(!r){ box.innerHTML=''; return; }

  if(r.state==='ask') box.innerHTML=`<div class="editing" style="flex-wrap:wrap"><span>Backup van ${dateShort(r.data.exportedAt.slice(0,10))}: ${r.data.visits.length} bezoeken, ${r.data.expenses.length} kosten, ${r.data.clients.length} klanten. Bestaande gegevens blijven staan; dezelfde items worden overschreven.</span><span style="display:flex;gap:6px"><button class="btn sm" type="button" id="rsGo">Terugzetten</button><button class="btn sm ghost" type="button" id="rsNo">Annuleren</button></span></div>`;

  else if(r.state==='run') box.textContent=`Bezig met terugzetten… ${r.done} / ${r.total}`;

  if($('rsGo')) $('rsGo').onclick=runRestore;

  if($('rsNo')) $('rsNo').onclick=()=>{ S.restore=null; renderBackupInfo(); };

}

$('bkDown').addEventListener('click',async()=>{

  const btn=$('bkDown'); btn.disabled=true; btn.textContent='Backup maken…';

  try{

    const months=new Set(S.index); Object.keys(S.visits).forEach(m=>{ if(monthHasData(m)) months.add(m); });

    const visits=[], expenses=[];

    for(const ym of [...months].sort()){

      const [vs,xs]=await Promise.all([S.db.collection(`months/${ym}/visits`).get(), S.db.collection(`months/${ym}/expenses`).get()]);

      vs.docs.forEach(d=>visits.push({ym,id:d.id,...d.data()})); xs.docs.forEach(d=>expenses.push({ym,id:d.id,...d.data()}));

    }

    const clients=[...S.clients.values()].map(c=>({...c}));

    const data={app:'sweetnails-salonboek',version:1,exportedAt:new Date().toISOString(),months:[...months].sort(),visits,expenses,clients};

    if(await saveFile(`Sweet Nails backup ${todayStr()}.json`, JSON.stringify(data,null,1))){

      await S.db.doc('meta/backup').set({lastAt:data.exportedAt}); toast('Backup bewaard op je toestel');

    }

  }catch(e){ console.error(e); toast('Backup maken is niet gelukt. Probeer opnieuw.',true); }

  finally{ btn.disabled=false; btn.textContent='Backup downloaden'; }

});

$('bkFile').addEventListener('change',async e=>{

  const f=e.target.files[0]; e.target.value=''; if(!f) return;

  try{

    const data=JSON.parse(await f.text());

    if(data.app!=='sweetnails-salonboek'||!Array.isArray(data.visits)) throw new Error('bad');

    data.expenses=data.expenses||[]; data.clients=data.clients||[];

    S.restore={state:'ask',data}; renderBackupInfo();

  }catch(err){ toast('Dit is geen geldig backupbestand van het Salonboek.',true); }

});

async function runRestore(){

  const data=S.restore.data; const jobs=[];

  const strip=o=>{ const c={...o}; delete c.id; delete c.ym; return c; };

  data.visits.forEach(v=>jobs.push(()=>S.db.doc(`months/${v.ym}/visits/${v.id}`).set(strip(v))));

  data.expenses.forEach(x=>jobs.push(()=>S.db.doc(`months/${x.ym}/expenses/${x.id}`).set(strip(x))));

  data.clients.forEach(c=>jobs.push(()=>S.db.doc(`clients/${c.id}`).set(strip(c))));

  const months=[...new Set(S.index.concat(data.months||[], data.visits.map(v=>v.ym), data.expenses.map(x=>x.ym)))].sort();

  jobs.push(()=>S.db.doc('meta/index').set({months}));

  S.restore={state:'run',data,done:0,total:jobs.length}; renderBackupInfo();

  try{

    for(const j of jobs){ await j(); S.restore.done++; if(S.restore.done%10===0) renderBackupInfo(); }

    toast('Backup teruggezet'); S.restore=null;

  }catch(e){ toast(`Terugzetten gestopt na ${S.restore.done} van ${S.restore.total} items. Probeer opnieuw; wat al terugstaat blijft staan.`,true); S.restore=null; }

  renderAll();

}

let yearUnsubs=[];

function subscribeYear(){

  yearUnsubs.forEach(u=>{ try{u();}catch(e){} }); yearUnsubs=[];

  if(!S.db) return;

  for(let m=1;m<=12;m++){

    const ym=S.year+'-'+pad(m);

    yearUnsubs.push(S.db.collection(`months/${ym}/visits`).onSnapshot(s=>{ S.visits[ym]=s.docs.map(d=>({id:d.id,ym,...d.data()})); S.loaded.add('v'+ym); scheduleRender(); }, onDbErr));

    yearUnsubs.push(S.db.collection(`months/${ym}/expenses`).onSnapshot(s=>{ S.expenses[ym]=s.docs.map(d=>({id:d.id,ym,...d.data()})); S.loaded.add('x'+ym); scheduleRender(); }, onDbErr));

  }

}

async function boot(){

  renderTabs(); renderMenu(); syncInputs();

  const h=(location.hash||'').slice(1); if(TABS.some(t=>t.id===h)) S.tab=h;

  go(S.tab);

  const c=window.claude;

  let db=null, dl=null;

  try{ [db,dl]=await Promise.all([c.use('db'), c.use('downloads').catch(()=>null)]); }catch(e){}

  S.dl=dl;

  if(!db){ S.dbState='none'; setBanner('<strong>De opslag is niet bereikbaar.</strong> Controleer of je aangemeld bent bij Claude en open het Salonboek opnieuw via de link.'); renderAll(); return; }

  S.db=db; S.dbState='ok';

  db.collection('clients').onSnapshot(s=>{ S.clients=new Map(s.docs.map(d=>[d.id,{id:d.id,...d.data()}])); S.clientsLoaded=true; scheduleRender(); }, onDbErr);

  subscribeYear(); renderAll();

}
